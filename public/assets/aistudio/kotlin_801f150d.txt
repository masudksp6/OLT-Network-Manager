        super.onCreate(savedInstanceState)
        setContent {
            ISPAppTheme {
                MainApp()
            }
        }
    }

@Composable
fun MainApp() {
    val navController = rememberNavController()
    val viewModel: MainViewModel = viewModel()

    // নেভিগেশন আইটেমের লিস্ট
    val navItems = listOf(
        BottomNavItem("pon", "পন পোর্ট", Icons.Default.Router),
        BottomNavItem("dashboard", "হোম", Icons.Default.Home),
        BottomNavItem("onu", "ONU", Icons.Default.DeviceHub),
        BottomNavItem("olt", "OLT", Icons.Default.Dns),
        BottomNavItem("more", "আরও", Icons.Default.AddCircle)
    )

    Scaffold(
        bottomBar = {
            NavigationBar(containerColor = androidx.compose.ui.graphics.Color.White) {
                val navBackStackEntry by navController.currentBackStackEntryAsState()
                val currentDestination = navBackStackEntry?.destination
                
                navItems.forEach { item ->
                    val selected = currentDestination?.hierarchy?.any { it.route == item.route } == true
                    NavigationBarItem(
                        icon = { Icon(item.icon, contentDescription = item.label) },
                        label = { Text(item.label, fontSize = 10.sp) },
                        selected = selected,
                        onClick = {
                            navController.navigate(item.route) {
                                popUpTo(navController.graph.findStartDestination().id) { saveState = true }
                                launchSingleTop = true
                                restoreState = true
                            }
                        }
                    )
                }
            }
        }
    ) { paddingValues ->
        NavHost(
            navController = navController,
            startDestination = "pon",
            modifier = Modifier.padding(paddingValues)
        ) {
            composable("pon") { PonScreen(viewModel) }
            composable("dashboard") { DashboardScreen(viewModel) }
            composable("onu") { OnuListScreen(viewModel) }
            composable("olt") { OltListScreen(viewModel) }
        }
    }
}

data class BottomNavItem(val route: String, val label: String, val icon: ImageVector)

🚀 অ্যাপটি রান করার নিয়ম

১. Android Studio খুলুন এবং উপরের সব ধাপ অনুযায়ী ফাইলগুলো তৈরি করুন (বা রিপ্লেস করুন)।
২. উপরে ডানদিকে "Sync Project with Gradle Files" আইকনে ক্লিক করুন।
৩. একটি ইমুলেটর (Emulator) বা রিয়েল ফোন কানেক্ট করুন।
৪. উপরে Run (▶️) বাটনে ক্লিক করুন।

💡 এই অ্যাপে যা যা কাজ করবে:

· পন পোর্ট স্ক্রিন: ফিল্টার বাটন (All, Optimal, Down) ক্লিক করলে ডেটা ফিল্টার হবে।
· হোম (ড্যাশবোর্ড): প্রতি ৩ সেকেন্ডে CPU, RAM, তাপমাত্রার মান নিজে নিজে পরিবর্তন হবে (লাইভ সিমুলেশন)।
· ONU লিস্ট: লাইভ সার্চ (SN বা নাম দিয়ে) এবং স্ট্যাটাস ফিল্টার কাজ করবে।
· OLT ম্যানেজমেন্ট: + বাটনে ক্লিক করলে পপ-আপ আসবে, সেখানে OLT অ্যাড করতে পারবেন। এডিট ও ডিলিট বাটনও কাজ করবে।

⚠️ একটি গুরুত্বপূর্ণ নোট:

জেটপ্যাক কম্পোজে (Jetpack Compose) বাংলা ফন্ট রেন্ডারিং অনেক ক্ষেত্রে ডিভাইসের সিস্টেম ফন্টের উপর নির্ভর করে। যদি কোনো কারণে বাংলা লেখা ভেঙে যায় বা বক্স আসে, তবে res/font ফোল্ডারে একটি বাংলা ফন্ট (যেমন: HindSiliguri-Regular.ttf) যোগ করে Type.kt-এ সেটি সেট করতে হবে। তবে আধুনিক বেশিরভাগ অ্যান্ড্রয়েড ফোনে (API 26+) বাংলা ডিফল্টভাবেই সাপোর্ট করে।
